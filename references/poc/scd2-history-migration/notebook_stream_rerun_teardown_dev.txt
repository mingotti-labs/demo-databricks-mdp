                     'change_ts', CAST(l.end_date AS TIMESTAMP), 'is_delete', true))
      ELSE array(
        named_struct('asset_id', l.asset_id, 'asset_name', l.asset_name, 'asset_owner', l.asset_owner,
                     'change_ts', CAST(l.start_date AS TIMESTAMP), 'is_delete', false))

);
